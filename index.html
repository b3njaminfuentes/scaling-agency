export function calculateSNA(s: Point, n: Point, a: Point): number {
  // Calculate angle between Sella-Nasion and Nasion-Point A
  const v1 = { x: s.x - n.x, y: s.y - n.y };
  const v2 = { x: a.x - n.x, y: a.y - n.y };
  
  const dot = v1.x * v2.x + v1.y * v2.y;
  const mag1 = Math.sqrt(v1.x * v1.x + v1.y * v1.y);
  const mag2 = Math.sqrt(v2.x * v2.x + v2.y * v2.y);
  
  const angleRad = Math.acos(dot / (mag1 * mag2));
  return (angleRad * 180) / Math.PI;
}
import { NextResponse } from 'next/server';
import { createMiddlewareClient } from '@supabase/auth-helpers-nextjs';

export async function middleware(req) {
  const res = NextResponse.next();
  const supabase = createMiddlewareClient({ req, res });
  
  const { data: { session } } = await supabase.auth.getSession();
  // Multi-tenant isolation at Edge
  if (session) {
    req.headers.set('x-tenant-id', session.user.app_metadata.tenant_id);
  }
  return res;
}
