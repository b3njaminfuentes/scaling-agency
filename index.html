export async function resolveClinicByHost(host: string): Promise<ResolvedClinic> {
  const cached = domainCache.get(cleanHost);
  if (cached && cached.expiresAt > Date.now()) return cached.clinic;

  const { data: clinic } = await supabase
    .from('Clinic')
    .select('id, name, slug, plan')
    .eq('domain', host)
    .single();

  // Cada consulta queda estrictamente aislada por clinicId
  return clinic;
}
